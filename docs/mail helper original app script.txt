  Auth.assertAllowedUser_();

  const userEmail = Session.getEffectiveUser().getEmail();
  const userConfig = Data.getUserConfig_(userEmail);
  
  const cfg = Data.getControllerConfig_();
  const allEvents = Data.getEvents_();

  // REMOVED: Auth.assertEventAccess_(payload.eventId);
  
  // Filter events based on user's allowed events
  let events = allEvents;
  if (userConfig && userConfig.allowedEventIds && userConfig.allowedEventIds.length > 0) {
    events = allEvents.filter(e => userConfig.allowedEventIds.includes(e.eventId));
  }

  if (events.length === 0) {
    throw new Error("Nemáte přístup k žádné události. Kontaktujte správce.");
  }

  // Determine default event: user's default > global config > first available
  let defaultEventId = null;
  if (userConfig && userConfig.defaultEventId) {
    defaultEventId = userConfig.defaultEventId;
  } else {
    defaultEventId = cfg.DEFAULT_EVENT_ID || (events[0] ? events[0].eventId : null);
  }

   const event = events.find(x => x.eventId === defaultEventId) || events[0];
  const children = Data.getChildren_(event);
  const documentTypes = Data.getDocumentTypes_(event.eventId);

  return {
    appTitle: cfg.APP_TITLE || "Camp Mail Helper",
    userEmail: userEmail,
    events,
    selectedEvent: event,
    children,
    documentTypes
  };

}

function api_buildReplyPreview(payload) {
  Auth.assertAllowedUser_();

  if (!payload || !payload.eventId || !payload.childName) {
    throw new Error("Neplatný požadavek (chybí eventId/childName).");
  }

  Auth.assertEventAccess_(payload.eventId);

  const events = Data.getEvents_();
  const event = events.find(x => x.eventId === payload.eventId);
  if (!event) throw new Error("Událost nenalezena: " + payload.eventId);

  const cfg = Data.getControllerConfig_();
  const questionnaireUrl = event.questionnaireUrl || String(cfg.QUESTIONNAIRE_URL || "").trim();

  const child = Data.findChildByName_(event, payload.childName);
  if (!child) throw new Error("Dítě nenalezeno: " + payload.childName);

  const documentTypes = Data.getDocumentTypes_(event.eventId);
  const actions = payload.attachmentActions || [];

  const simulatedChildData = { ...child };

  actions.forEach(a => {
    const targetChildName = String(a.childName || payload.childName || "").trim();

    if (targetChildName !== String(payload.childName || "").trim()) return;

    if (a.action !== "IGNORE") {
      const docType = documentTypes.find(d => d.documentKey === a.action);
      if (docType && docType.columnName) {
        simulatedChildData[docType.columnName] = docType.expectedValue;
      }
    }
  });

  const appDocType = documentTypes.find(d => d.documentKey === "APPLICATION");
  const isFirstTimeApplication = appDocType &&
    actions.some(a =>
      a.action === "APPLICATION" &&
      String(a.childName || payload.childName || "").trim() === String(payload.childName || "").trim()
    ) &&
    String(child[appDocType.columnName] || "").trim().toLowerCase() !==
      String(appDocType.expectedValue || "yes").trim().toLowerCase();

  const userEmail = Session.getEffectiveUser().getEmail();
  const signature = Data.getUserSignature_(userEmail);

  const text = GmailSvc.buildSingleReplyTextCz_({
    childName: child.childName,
    childData: simulatedChildData,
    documentTypes,
    questionnaireUrl,
    note: String(payload.note || "").trim(),
    signature,
    isFirstTimeApplication
  });

  return { replyText: text };
}

function api_executeCustomActions(payload) {
  Auth.assertAllowedUser_();

  if (!payload || !payload.eventId || !payload.messageId) {
    throw new Error("Neplatný požadavek (chybí eventId/messageId).");
  }

  Auth.assertEventAccess_(payload.eventId);

  const events = Data.getEvents_();
  const event = events.find(x => x.eventId === payload.eventId);
  if (!event) throw new Error("Událost nenalezena: " + payload.eventId);

  const msg = GmailSvc.getMessageObject_(payload.messageId);
  let saved = { savedFiles: [] };

  // Action 1: Save attachments
  if (payload.actions.saveAttachments) {
    if (!payload.childName) {
      throw new Error("Pro uložení příloh je potřeba vybrat dítě.");
    }

    const attachments = GmailSvc.listAttachments_(msg);
    const documentTypes = Data.getDocumentTypes_(event.eventId);

    const actionMap = {};
    (payload.attachmentActions || []).forEach(a => {
      actionMap[String(a.attachmentId)] = {
        action: a.action,
        childName: String(a.childName || payload.childName || "").trim()
      };
    });

    const docTypeCounter = {};

    attachments.forEach(att => {
      const mapped = actionMap[String(att.attachmentId)] || {
        action: "IGNORE",
        childName: String(payload.childName || "").trim()
      };

      const action = mapped.action || "IGNORE";
      if (action === "IGNORE") return;

      const targetChildName = mapped.childName || String(payload.childName || "").trim();
      const targetChild = Data.findChildByName_(event, targetChildName);
      if (!targetChild) {
        throw new Error("Dítě pro přílohu nenalezeno: " + targetChildName);
      }

      const docType = documentTypes.find(d => d.documentKey === action);
      if (!docType || !docType.filenameSuffix) return;

      const childFolder = DriveSvc.getChildFolder_(event.rootFolderId, targetChild.folderName);

      const blob = att.blob;
      const ext = DriveSvc.getSafeExtension_(blob.getName());
      const base = targetChild.childName;

      const counterKey = `${targetChild.childName}__${action}`;
      if (!docTypeCounter[counterKey]) {
        docTypeCounter[counterKey] = 0;
      }
      docTypeCounter[counterKey]++;

      const suffix = docTypeCounter[counterKey] > 1 ? `_${docTypeCounter[counterKey]}` : "";
      const newName = `${base}_${docType.filenameSuffix}${suffix}${ext}`;

      const file = childFolder.createFile(blob).setName(newName);

      saved.savedFiles.push({
        name: newName,
        fileId: file.getId(),
        documentKey: action,
        childName: targetChild.childName,
        rowIndex: targetChild.rowIndex
      });

      if (!saved.documents) saved.documents = {};
      if (!saved.documentsByChild) saved.documentsByChild = {};

      saved.documents[action] = true;

      if (!saved.documentsByChild[targetChild.rowIndex]) {
        saved.documentsByChild[targetChild.rowIndex] = {
          childName: targetChild.childName,
          rowIndex: targetChild.rowIndex,
          documents: {}
        };
      }

      saved.documentsByChild[targetChild.rowIndex].documents[action] = true;
    });
  }

  // Action 2: Send reply
  if (payload.actions.sendReply) {
    if (!payload.replyText || !payload.replyText.trim()) {
      throw new Error("Pro odeslání odpovědi je potřeba zadat text.");
    }
    GmailSvc.replyToMessage_(payload.messageId, payload.replyText);
  }

  // Action 3: Move email
  if (payload.actions.moveEmail) {
    GmailSvc.moveToDone_(payload.messageId, event.doneLabelName);
  }

  // Action 4: Update table
  if (payload.actions.updateTable && saved.documentsByChild) {
    const documentTypes = Data.getDocumentTypes_(event.eventId);

    Object.keys(saved.documentsByChild).forEach(rowKey => {
      const item = saved.documentsByChild[rowKey];
      const updates = {};

      Object.keys(item.documents || {}).forEach(docKey => {
        const docType = documentTypes.find(d => d.documentKey === docKey);
        if (docType && docType.columnName) {
          updates[docType.columnName] = docType.expectedValue;
        }
      });

      Data.updateChildDocFlags_(event, Number(item.rowIndex), updates);
    });
  }

  return {
    ok: true,
    saved,
    actionsExecuted: payload.actions
  };
}

const Auth = (() => {

  function getUserEmail_() {
    try {
      const e1 = Session.getActiveUser().getEmail();
      if (e1) return e1;
    } catch (err) {}

    try {
      const e2 = Session.getEffectiveUser().getEmail();
      if (e2) return e2;
    } catch (err) {}

    throw new Error(
      "Nepodařilo se zjistit email uživatele. Zkontrolujte oauthScopes (userinfo.email)."
    );
  }

  function assertAllowedUser_() {
    const email = getUserEmail_();
    const allowed = Data.getAllowedUsers_();
    const ok = allowed.some(u => (u.email || "").toLowerCase() === email.toLowerCase());

    if (!ok) {
      throw new Error(
        "Nemáte přístup k této aplikaci. Požádejte správce, aby vás přidal do listu Users. " +
        `(${email})`
      );
    }
  }

  function assertEventAccess_(eventId) {
    const email = getUserEmail_();
    const userConfig = Data.getUserConfig_(email);
    
    if (!userConfig) {
      throw new Error("Uživatel nenalezen v konfiguraci.");
    }

    // If no allowed events defined, allow all (backward compatibility)
    if (!userConfig.allowedEventIds || userConfig.allowedEventIds.length === 0) {
      return;
    }

    // Check if user has access to this event
    const hasAccess = userConfig.allowedEventIds.includes(eventId);
    if (!hasAccess) {
      throw new Error(`Nemáte přístup k události "${eventId}". Kontaktujte správce.`);
    }
  }

return { getUserEmail_, assertAllowedUser_, assertEventAccess_ };

const Data = (() => {

  function controllerSs_() {
    return SpreadsheetApp.getActiveSpreadsheet();
  }

  function sheetByName_(ss, name) {
    const sh = ss.getSheetByName(name);
    if (!sh) throw new Error(`Chybí list "${name}" v controller spreadsheetu.`);
    return sh;
  }

  function readTable_(sheet) {
    const values = sheet.getDataRange().getValues();
    if (!values.length) return [];
    const headers = values[0].map(h => String(h || "").trim());
    const rows = [];
    for (let i = 1; i < values.length; i++) {
      const obj = {};
      headers.forEach((h, idx) => obj[h] = values[i][idx]);
      obj._rowIndex = i + 1; // 1-based
      rows.push(obj);
    }
    return rows;
  }

  function normalizeMultiline_(s) {
    let t = String(s || "");

    // Trim whitespace
    t = t.trim();

    // Remove surrounding quotes if user pasted "...."
    if ((t.startsWith('"') && t.endsWith('"')) || (t.startsWith("'") && t.endsWith("'"))) {
      t = t.slice(1, -1);
    }

    // Convert literal \n into real newlines
    // (user sometimes writes \n inside a cell)
    t = t.replace(/\\n/g, "\n");

    // Normalize CRLF
    t = t.replace(/\r\n/g, "\n").replace(/\r/g, "\n");

    return t.trim();
  }

  /**
   * Supports BOTH formats:
   * A) Vertical:
   *   Key | Value
   *   APP_TITLE | ...
   * B) Wide (your current):
   *   DEFAULT_EVENT_ID | APP_TITLE | QUESTIONNAIRE_URL
   *   SUMMER_2026      | ...       | https://...
   */
  function getControllerConfig_() {
    const sh = sheetByName_(controllerSs_(), "Config");
    const values = sh.getDataRange().getValues();
    if (!values.length) return {};

    const header = values[0].map(x => String(x || "").trim());
    const out = {};

    // Case A: first row includes Key/Value headers
    const keyIdx = header.findIndex(h => h.toLowerCase() === "key");
    const valIdx = header.findIndex(h => h.toLowerCase() === "value");
    if (keyIdx >= 0 && valIdx >= 0) {
      for (let r = 1; r < values.length; r++) {
        const k = String(values[r][keyIdx] || "").trim();
        if (!k) continue;
        out[k] = String(values[r][valIdx] || "").trim();
      }
      return out;
    }

    // Case B: wide format => take the first non-empty data row
    let dataRow = null;
    for (let r = 1; r < values.length; r++) {
      const any = values[r].some(v => String(v || "").trim() !== "");
      if (any) { dataRow = values[r]; break; }
    }
    if (!dataRow) return {};

    header.forEach((k, i) => {
      const key = String(k || "").trim();
      if (!key) return;
      out[key] = String(dataRow[i] || "").trim();
    });

    return out;
  }

  function getEvents_() {
    const sh = sheetByName_(controllerSs_(), "Events");

function listInboxMessages_(maxCount) {
  maxCount = Math.max(1, Math.min(Number(maxCount || 25), 200));

  // Message view query:
  // - in:inbox = only inbox
  // - -from:me = exclude your sent replies from the stream
  const q = 'in:inbox -from:me';

  const res = Gmail.Users.Messages.list('me', {
    q,
    maxResults: maxCount
  });

  const ids = (res.messages || []).map(m => m.id);
  const out = [];

  ids.forEach(id => {
    const msg = Gmail.Users.Messages.get('me', id, { format: 'metadata' });

    const headers = {};
    (msg.payload.headers || []).forEach(h => headers[h.name.toLowerCase()] = h.value);

    out.push({
      messageId: id,
      threadId: msg.threadId || '',
      from: headers['from'] || '',
      subject: headers['subject'] || '',
      date: (headers['date'] ? new Date(headers['date']).toISOString() : ''),
      snippet: (msg.snippet || '').slice(0, 200)
    });
  });

  // newest first (API already returns newest, but keep it deterministic)
  out.sort((a, b) => (b.date || '').localeCompare(a.date || ''));
  return out;

  function getSafeExtension_(originalName) {
    const name = String(originalName || "");
    const idx = name.lastIndexOf(".");
    if (idx < 0) return "";
    const ext = name.slice(idx).trim();
    if (ext.length > 10) return "";
    return ext;
  }

  function getOrCreateFolder_(parentFolderId, folderName) {
    if (!parentFolderId) throw new Error("Chybí parentFolderId.");
    if (!folderName) throw new Error("Chybí folderName.");

    const parent = DriveApp.getFolderById(parentFolderId);
    const it = parent.getFoldersByName(folderName);
    if (it.hasNext()) return it.next();
    return parent.createFolder(folderName);
  }

  function cleanupOldFiles_(folder, options) {
  options = options || {};
  const maxAgeHours = Number(options.maxAgeHours || 24);
  const keepNewest = Number(options.keepNewest || 50);

  const now = Date.now();
  const cutoff = now - (maxAgeHours * 60 * 60 * 1000);

  // Collect files
  const files = [];
  const it = folder.getFiles();
  while (it.hasNext()) {
    const f = it.next();
    files.push(f);
  }

  // Sort newest first
  files.sort((a, b) => b.getLastUpdated().getTime() - a.getLastUpdated().getTime());

  // Keep N newest always
  const keepSet = new Set(files.slice(0, keepNewest).map(f => f.getId()));

  // Trash old ones + anything beyond keepNewest (optional)
  files.forEach((f, idx) => {
    const updated = f.getLastUpdated().getTime();
    const isOld = updated < cutoff;
    const beyondCap = idx >= keepNewest;

    if (isOld || beyondCap) {
      try { f.setTrashed(true); } catch (e) {}
    }
  });
}


  return { getChildFolder_, getSafeExtension_, getOrCreateFolder_, cleanupOldFiles_  };

  function getSortedEmails(){
    const sort = document.getElementById('emailSort')?.value || 'oldest';
    const arr = [...STATE.emails];

    arr.sort((a, b) => {
      const da = new Date(a.date || 0).getTime();
      const db = new Date(b.date || 0).getTime();
      return sort === 'oldest' ? da - db : db - da;
    });

    return arr;
  }

  function getNextEmailId(currentId){
    const sorted = getSortedEmails();
    const idx = sorted.findIndex(e => e.messageId === currentId);
    if (idx < 0) return sorted[0]?.messageId || null;
    return sorted[idx + 1]?.messageId || sorted[idx - 1]?.messageId || null;
  }

  function detectChildFromEmail(detail){
    const text = normalize((detail.subject || '') + ' ' + (detail.bodySnippet || ''));
    let best = null;

    STATE.children.forEach(c => {
      const name = normalize(c.childName);
      if (name && text.includes(name)){
        if (!best || name.length > normalize(best).length) best = c.childName;
      }
    });

    return best;
  }

  async function executeSelectedActions(){
    const status = document.getElementById('processStatus');

    if (!STATE.selectedEmail){
      status.textContent = 'Vyberte email.';
      return;
    }

    const currentEmail = STATE.selectedEmail;
    const currentEmailId = currentEmail.messageId;

    const actions = {
      saveAttachments: document.getElementById('actionSaveAttachments').checked,
      sendReply: document.getElementById('actionSendReply').checked,
      moveEmail: document.getElementById('actionMoveEmail').checked,
      updateTable: document.getElementById('actionUpdateTable').checked
    };

    if (actions.saveAttachments || actions.updateTable) {
      const childName = document.getElementById('childSelect').value;
      if (!childName) {
        status.textContent = 'Pro uložení příloh nebo aktualizaci tabulky vyberte dítě.';
        return;
      }
    }

    if (actions.sendReply) {
      const replyText = document.getElementById('replyEdit').value || '';
      if (!replyText.trim()) {
        status.textContent = 'Pro odeslání odpovědi zadejte text.';
        return;
      }
    }

const mainChildName = document.getElementById('childSelect').value;

const attachmentActions = Object.keys(STATE.attachmentActions).map(k => ({
  attachmentId: k,
  action: STATE.attachmentActions[k],
  childName: STATE.attachmentChildren[k] || mainChildName
}));

    const payload = {
      eventId: STATE.selectedEvent.eventId,
      messageId: currentEmailId,
      childName: document.getElementById('childSelect').value,
      replyText: document.getElementById('replyEdit').value,
      attachmentActions,
      actions
    };

    const nextId = getNextEmailId(currentEmailId);

    STATE.emails = STATE.emails.map(e => {
      if (e.messageId === currentEmailId) return Object.assign({}, e, { _processing: true });
      return e;
    });

    renderEmailList();
    status.textContent = 'Zpracování spuštěno. Můžete pokračovat dalším emailem.';

    if (nextId && nextId !== currentEmailId) {
      showEmailDetail(nextId);
    } else {
      STATE.selectedEmail = null;
      STATE.selectedEmailId = null;
      document.getElementById('emailDetailEmpty').style.display = 'block';
      document.getElementById('emailDetail').style.display = 'none';
    }

    api('api_executeCustomActions', payload)
      .then(result => {
        const executed = [];
        if (actions.saveAttachments && result.saved.savedFiles.length > 0) executed.push('přílohy uloženy');
        if (actions.sendReply) executed.push('odpověď odeslána');
        if (actions.moveEmail) executed.push('email přesunut');
        if (actions.updateTable && result.saved.documents && Object.keys(result.saved.documents).length > 0) {
          executed.push('tabulka aktualizována');
        }

        STATE.emails = STATE.emails.filter(e => e.messageId !== currentEmailId);
        delete STATE.emailDetailsById[currentEmailId];

        renderEmailList();
        setStatus('inboxStatus', executed.length ? `Hotovo: ${executed.join(', ')}.` : 'Hotovo.');
      })
      .catch(e => {
        STATE.emails = STATE.emails.map(mail => {
          if (mail.messageId === currentEmailId) return Object.assign({}, mail, { _processing: false });
          return mail;
        });

        renderEmailList();
        setStatus('inboxStatus', 'Chyba při zpracování emailu: ' + (e && e.message ? e.message : e));
      });
  }

  function toggleTheme(){
    const current = document.documentElement.classList.contains('light-mode') ? 'light' : 'dark';
    const newTheme = current === 'light' ? 'dark' : 'light';

    if (newTheme === 'light') {
      document.documentElement.classList.add('light-mode');
    } else {
      document.documentElement.classList.remove('light-mode');
    }

    localStorage.setItem('theme', newTheme);
    updateThemeLabel(newTheme);
  }

  function updateThemeLabel(theme) {
    const label = document.getElementById('themeLabel');
    if (label) {
      label.textContent = theme === 'dark' ? '☀️' : '🌙';
    }
  }

  function initTheme(){
    const savedTheme = localStorage.getItem('theme') || 'dark';
    if (savedTheme === 'light') {
      document.documentElement.classList.add('light-mode');
    }
    updateThemeLabel(savedTheme);
  }
